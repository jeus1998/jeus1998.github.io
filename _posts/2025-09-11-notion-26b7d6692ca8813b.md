```bash
# mysql 같은 경우 환경변수로 비밀번호를 지정하지 않고 시작하면 에러가 발생한다.
docker run -e MYSQL_ROOT_PASSWORD=[REDACTED] -d -p 3306:3306 mysql

---------------------------------------------------------------------------------
# 1. 컨테이너 내부 접속하기
docker exec -it [컨테이너 ID] bash

# 2. 비밀번호(환경 변수) 출력하기
echo $MYSQL_ROOT_PASSWORD
---------------------------------------------------------------------------------

# 컨테이너 mysql 바로 접속하기
docker exec -it [컨테이너 ID] mysql -u root -p 
```
