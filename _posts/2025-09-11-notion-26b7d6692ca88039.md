{% raw %}# [프로그래머스] 섬 연결하기

**탐욕법(Greedy), LEVEL3** 

### 크루스칼 알고리즘 풀이

```java
import java.util.*;
class Solution {
    static int [] group;
    static int answer = 0;
    public int solution(int n, int[][] costs) {
        init(n);
        core(n, costs);
        return answer;
    }
    static void init(int n) {
        group = new int [n];
        for(int i = 1; i < n; i++) {
            group[i] = i;
        }
    }
    static void core(int n, int[][] costs) {
        Arrays.sort(costs, Comparator.comparingInt(o -> o[2]));
        int cnt = 0; // 연결된 간선의 개수
        int idx = 0;
        while(cnt < n - 1) {
            int [] cur = costs[idx++];
            if(find(cur[0]) == find(cur[1])) continue;
            union(cur[0], cur[1]);
            answer += cur[2];
            cnt++; // 간선 추가 
        }
    }
    static int find(int x) {
        if(group[x] == x) return x;
        return group[x] = find(group[x]);
    }
    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if(x > y) group[x] = y;
        else group[y] = x;
    }
}
```

### 프림 알고리즘 풀이

```java
import java.util.*;
class Solution {
    static int answer = 0;
    static ArrayList<int[]>[] graph;
    public int solution(int n, int[][] costs) {
        init(n, costs);
        core(n);
        return answer;
    }
    static void init(int n, int [][] costs) {
        graph = new ArrayList[n];
        // 그래프 초기화
        for(int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        // 그래프 (양방향)연결
        for(int [] cur : costs) {
           graph[cur[0]].add(new int [] {cur[1], cur[2]});
           graph[cur[1]].add(new int [] {cur[0], cur[2]}); 
        }
    }
    static void core(int n) {
        int cnt = 0; // 연결된 간선의 개수
        boolean [] visited = new boolean [n]; // 방문 체크용
        visited[0] = true; // 0번으로 시작
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(o -> o[1]));
        for(int [] start : graph[0]) {
            pq.add(start);
        }
        while(cnt < n - 1) {
            int [] cur = pq.poll();
            if(visited[cur[0]]) continue;
            visited[cur[0]] = true;
            answer += cur[1];
            cnt++;
            for(int [] next : graph[cur[0]]) {
                if(visited[next[0]]) continue;
                pq.add(next);
            }
        }
    }
}
```
